Teaching pseudocode; all handlers hold the same mutex.
retire(i):
  require state[i] == indexed
  remove i from index
  state[i] = retired
  check(i)
release(i):
  require r[i] > 0 and state[i] != freed
  r[i] = r[i] - 1
  check(i)
check(i):
  if state[i] == retired and r[i] == 0:
    free storage(i)
    state[i] = freed
